# STEP 1: Import Required Libraries

import pandas as pd

import numpy as np

import matplotlib.pyplot as plt

import seaborn as sns

from sklearn.datasets import load_iris

from sklearn.preprocessing import StandardScaler

from sklearn.cluster import KMeans, AgglomerativeClustering

from sklearn.decomposition import PCA

from sklearn.metrics import silhouette_score

from scipy.cluster.hierarchy import linkage, dendrogram

# STEP 2: Load Iris Dataset

iris = load_iris()

df = pd.DataFrame(

   iris.data,

   columns=iris.feature_names

)

print("First Five Records:")

print(df.head())

print("\nDataset Shape:")

print(df.shape)

print("\nMissing Values:")

print(df.isnull().sum())

# STEP 3: Preprocess and Scale the Data

# Handle missing values

df = df.fillna(df.median(numeric_only=True))

# Standardize the features

scaler = StandardScaler()

X_scaled = scaler.fit_transform(df)

print("\nScaled Data (First Five Records):")

print(X_scaled[:5])

# STEP 4: Find Optimal K using Elbow Method

wcss = []

for k in range(1, 11):

   kmeans = KMeans(

       n_clusters=k,

       random_state=42,

       n_init=10

   )

   kmeans.fit(X_scaled)

wcss.append(kmeans.inertia_)

# Plot Elbow Graph

plt.figure(figsize=(8, 5))

plt.plot(

   range(1, 11),

wcss,

   marker='o',

   linewidth=2

)

plt.title("Elbow Method for Optimal K")

plt.xlabel("Number of Clusters (K)")

plt.ylabel("WCSS")

plt.xticks(range(1, 11))

plt.grid(True)

plt.tight_layout()

plt.show()

# STEP 5: Implement K-Means Clustering

# Select 3 clusters

kmeans = KMeans(

   n_clusters=3,

   random_state=42,

   n_init=10

)

kmeans_labels = kmeans.fit_predict(X_scaled)

df["KMeans_Cluster"] = kmeans_labels

print("\nK-Means Cluster Counts:")

print(

   df["KMeans_Cluster"]

   .value_counts()

   .sort_index()

)

# STEP 6: Visualize K-Means Clusters using PCA

pca = PCA(n_components=2)

X_pca = pca.fit_transform(X_scaled)

plt.figure(figsize=(8, 5))

plt.scatter(

   X_pca[:, 0],

   X_pca[:, 1],

   c=kmeans_labels,

   cmap="viridis",

   s=50

)

plt.title("K-Means Clustering of Iris Dataset")

plt.xlabel("Principal Component 1")

plt.ylabel("Principal Component 2")

plt.colorbar(label="Cluster")

plt.grid(True)

plt.tight_layout()

plt.show()

# STEP 7: Implement Hierarchical Clustering

hierarchical = AgglomerativeClustering(

   n_clusters=3,

   linkage="ward"

)

hierarchical_labels = hierarchical.fit_predict(X_scaled)

df["Hierarchical_Cluster"] = hierarchical_labels

print("\nHierarchical Cluster Counts:")

print(

   df["Hierarchical_Cluster"]

   .value_counts()

   .sort_index()

)

# STEP 8: Plot Dendrogram

Z = linkage(

   X_scaled,

   method="ward"

)

plt.figure(figsize=(12, 6))

dendrogram(

   Z,

   truncate_mode="lastp",

   p=30,

   show_leaf_counts=True

)

plt.title("Dendrogram - Hierarchical Clustering")

plt.xlabel("Data Points / Merged Groups")

plt.ylabel("Ward Linkage Distance")

plt.tight_layout()

plt.show()

# STEP 9: Visualize Hierarchical Clustering

plt.figure(figsize=(8, 5))

plt.scatter(

   X_pca[:, 0],

   X_pca[:, 1],

   c=hierarchical_labels,

   cmap="plasma",

   s=50

)

plt.title("Hierarchical Clustering of Iris Dataset")

plt.xlabel("Principal Component 1")

plt.ylabel("Principal Component 2")

plt.colorbar(label="Cluster")

plt.grid(True)

plt.tight_layout()

plt.show()

# STEP 10: Compare Both Clustering Algorithms

kmeans_silhouette = silhouette_score(

   X_scaled,

   kmeans_labels

)

hierarchical_silhouette = silhouette_score(

   X_scaled,

   hierarchical_labels

)

results = pd.DataFrame({

   "Algorithm": [

       "K-Means",

       "Hierarchical Clustering"

   ],

   "Number of Clusters": [

       3,

       3

   ],

   "Silhouette Score": [

       round(kmeans_silhouette, 3),

       round(hierarchical_silhouette, 3)

   ]

})

print("\n======================================")

print("CLUSTERING MODEL COMPARISON")

print("======================================")

print(results.to_string(index=False))

# STEP 11: Analyze Generated Clusters

print("\nK-Means Cluster Analysis:")

print(

   df.groupby("KMeans_Cluster")[

       iris.feature_names

   ].mean().round(2)

)

 

print("\nHierarchical Cluster Analysis:")

print(

   df.groupby("Hierarchical_Cluster")[

       iris.feature_names

   ].mean().round(2)

)

# STEP 12: Compare Clusters with Actual Iris Species

comparison = pd.crosstab(

   pd.Series(

       kmeans_labels,

       name="KMeans Cluster"

   ),

   pd.Series(

       iris.target_names[iris.target],

       name="Actual Species"

   )

)

print("\nK-Means Cluster vs Actual Species:")

print(comparison)

# STEP 13: Final Result

print("\n======================================")

print("PRACTICAL COMPLETED SUCCESSFULLY!")

print("======================================")

if kmeans_silhouette > hierarchical_silhouette:

   print("K-Means has a higher Silhouette Score.")

elif hierarchical_silhouette > kmeans_silhouette:

   print("Hierarchical Clustering has a higher Silhouette Score.")

else:

   print("Both algorithms have equal Silhouette Scores.")
