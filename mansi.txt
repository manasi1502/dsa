class SmartHomeAgent:
    def __init__(self):
        self.light = "OFF"
        self.ac = "OFF"
        self.fan = "OFF"
        self.alarm = "OFF"
    
    def perceive(self, temperature, light_level, motion):
        
        if temperature > 30:
            self.ac = "ON"
            self.fan = "ON"
        else:
            self.ac = "OFF"
            self.fan = "OFF"
        
        if light_level < 40:
            self.light = "ON"
        else:
            self.light = "OFF"
        
        if motion:
            self.alarm = "OFF"
        else:
            self.alarm = "ON"
   

 
    def display_status(self):
        print("\n----- SMART HOME STATUS -----")
        print("Light           :", self.light)
        print("Air Conditioner :", self.ac)
        print("Fan             :", self.fan)
        print("Security Alarm  :", self.alarm)

agent = SmartHomeAgent()

temperature = float(
    input("Enter temperature (°C): ")
)
light_level = int(
    input("Enter light intensity (0-100): ")
)
motion_input = input(
    "Is motion detected? (yes/no): "
).lower()
motion_detected = motion_input == "yes"

agent.perceive(
    temperature,
    light_level,
    motion_detected
)

agent.display_status()

class ChatbotAgent:

    def __init__(self):
        self.response = ""

    
    def perceive(self, message):

        message = message.lower()

        
        if "hello" in message or "hi" in message:
            self.response = "Hello! How can I help you?"

        
        elif "how are you" in message:
            self.response = "I am fine! Thank you for asking."

        
        elif "help" in message:
            self.response = "Sure! I can answer simple questions."

        
        elif "bye" in message:
            self.response = "Goodbye! Have a nice day."

        
        else:
            self.response = "Sorry, I don't understand that."

   
    def display_response(self):

        print("\n----- CHATBOT RESPONSE -----")
        print("Bot :", self.response)



agent = ChatbotAgent()


message = input("You: ")


agent.perceive(message)


agent.display_response()

class RecommendationAgent:

    def __init__(self):
        self.recommendation = ""

    
    def perceive(self, genre, rating):
        genre = genre.lower()

       
        if genre == "action" and rating >= 7:
            self.recommendation = "Recommended: Avengers: Endgame"

        
        elif genre == "comedy" and rating >= 7:
            self.recommendation = "Recommended: 3 Idiots"

        
        elif genre == "sci-fi" and rating >= 7:
            self.recommendation = "Recommended: Interstellar"

       
        elif genre == "romance" and rating >= 7:
            self.recommendation = "Recommended: The Notebook"

        
        elif rating < 7:
            self.recommendation = "Try movies with a rating of 7 or above."

        
        else:
            self.recommendation = "Sorry, no recommendation available."

   
    def display_recommendation(self):
        print("\n----- MOVIE RECOMMENDATION -----")
        print(self.recommendation)



agent = RecommendationAgent()

rating = float(input("Enter minimum preferred rating (0-10): "))


agent.perceive(genre, rating)


agent.display_recommendation()

import heapq


graph = {
    'S': {'A': 1, 'B': 2},
    'A': {'C': 2},
    'B': {'D': 2},
    'C': {'G': 2},
    'D': {'G': 10},
    'G': {}
}


heuristic = {
    'S': 4,
    'A': 4,
    'B': 2,
    'C': 2,
    'D': 1,
    'G': 0
}



def greedy_best_first(start, goal):
    pq = [(heuristic[start], start, [start])]
    visited = set()

    while pq:
        h, node, path = heapq.heappop(pq)

        if node in visited:
            continue

        visited.add(node)

        if node == goal:
            return path, len(visited)

        for neighbor in graph[node]:
            if neighbor not in visited:
                heapq.heappush(
                    pq,
                    (heuristic[neighbor], neighbor,
                     path + [neighbor])
                )

    return None, len(visited)



def a_star(start, goal):
    pq = [(heuristic[start], 0, start, [start])]
    g_cost = {start: 0}
    visited = set()

    while pq:
        f, g, node, path = heapq.heappop(pq)

        if g > g_cost.get(node, float('inf')):
            continue

        if node in visited:
            continue

        visited.add(node)

        if node == goal:
            return path, g, len(visited)

        for neighbor, cost in graph[node].items():
            new_g = g + cost

            if new_g < g_cost.get(neighbor, float('inf')):
                g_cost[neighbor] = new_g
                new_f = new_g + heuristic[neighbor]

                heapq.heappush(
                    pq,
                    (new_f, new_g, neighbor,
                     path + [neighbor])
                )

    return None, float('inf'), len(visited)



start = 'S'
goal = 'G'

print("Greedy Best First Search:")
path, count = greedy_best_first(start, goal)
print("Path:", path)
if path:
    cost = sum(graph[path[i]][path[i+1]]
               for i in range(len(path)-1))
    print("Total Cost:", cost)
print("Nodes Visited:", count)

print("\nA* Search:")
path, cost, count = a_star(start, goal)
print("Path:", path)
print("Total Cost:", cost)
print("Nodes Visited:", count)

import numpy as np

import pandas as pd

import matplotlib.pyplot as plt

from sklearn.datasets import fetch_california_housing

from sklearn.model_selection import train_test_split

from sklearn.linear_model import LinearRegression

from sklearn.preprocessing import PolynomialFeatures

from sklearn.pipeline import make_pipeline

from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score

# 1. Load Dataset

housing = fetch_california_housing()

df = pd.DataFrame(

   housing.data,

   columns=housing.feature_names

)

df["MedHouseValue"] = housing.target

print("First 5 Records:")

print(df.head())

print("\nDataset Shape:")

print(df.shape)

print("\nMissing Values:")

print(df.isnull().sum())

# 2. Select Feature and Target

X = df[["MedInc"]]

y = df["MedHouseValue"]

# 3. Train-Test Split

X_train, X_test, y_train, y_test = train_test_split(

   X,

   y,

   test_size=0.20,

   random_state=42

)

# 4. Linear Regression

linear_model = LinearRegression()

linear_model.fit(X_train, y_train)

y_pred_linear = linear_model.predict(X_test)

# 5. Linear Regression Evaluation

mse_linear = mean_squared_error(y_test, y_pred_linear)

mae_linear = mean_absolute_error(y_test, y_pred_linear)

r2_linear = r2_score(y_test, y_pred_linear)

# 6. Polynomial Regression

degree = 2

poly_model = make_pipeline(

   PolynomialFeatures(degree=degree),

   LinearRegression()

)

poly_model.fit(X_train, y_train)

y_pred_poly = poly_model.predict(X_test)

# 7. Polynomial Regression Evaluation

mse_poly = mean_squared_error(y_test, y_pred_poly)

mae_poly = mean_absolute_error(y_test, y_pred_poly)

r2_poly = r2_score(y_test, y_pred_poly)

# 8. Display Results

results = pd.DataFrame({

   "Model": [

       "Linear Regression",

       "Polynomial Regression"

   ],

   "MSE": [

       mse_linear,

       mse_poly

   ],

   "MAE": [

       mae_linear,

       mae_poly

   ],

   "R2 Score": [

       r2_linear,

      r2_poly

   ]

})

print("\n==============================")

print("REGRESSION MODEL COMPARISON")

print("==============================")

print(results.round(6))

# 9. Visualization

# Create DataFrame with correct feature name

X_plot = pd.DataFrame(

   np.linspace(

       X["MedInc"].min(),

       X["MedInc"].max(),

       500

   ),

   columns=["MedInc"]

)

# Predict values

y_linear_plot = linear_model.predict(X_plot)

y_poly_plot = poly_model.predict(X_plot)

# Plot graph

plt.figure(figsize=(10, 6))

plt.scatter(

   X_test["MedInc"],

   y_test,

   alpha=0.3,

label="Actual Data"

)

plt.plot(

   X_plot["MedInc"],

   y_linear_plot,

   label="Linear Regression",

   linewidth=2

)

plt.plot(

   X_plot["MedInc"],

   y_poly_plot,

   label="Polynomial Regression",

   linewidth=2

)

plt.xlabel("Median Income")

plt.ylabel("Median House Value")

plt.title("Linear vs Polynomial Regression")

plt.legend()

plt.grid(True)

plt.show()

# 10. Final Result

print("\nPractical Completed Successfully!")

if r2_poly > r2_linear:

   print("Polynomial Regression performs better.")

else:

   print("Linear Regression performs better.")

import pandas as pd

import numpy as np

import matplotlib.pyplot as plt

import seaborn as sns

from sklearn.cluster import KMeans, AgglomerativeClustering

from sklearn.decomposition import PCA

from sklearn.metrics import silhouette_score

from scipy.cluster.hierarchy import linkage, dendrogram

# STEP 2: Load Iris Dataset

iris = load_iris()

df = pd.DataFrame(

   iris.data,

   columns=iris.feature_names

)

print("First Five Records:")

print(df.head())

print("\nDataset Shape:")

print(df.shape)

print("\nMissing Values:")

print(df.isnull().sum())

# STEP 3: Preprocess and Scale the Data

# Handle missing values

df = df.fillna(df.median(numeric_only=True))

# Standardize the features

scaler = StandardScaler()

X_scaled = scaler.fit_transform(df)

print("\nScaled Data (First Five Records):")

print(X_scaled[:5])

# STEP 4: Find Optimal K using Elbow Method

wcss = []

for k in range(1, 11):

   kmeans = KMeans(

       n_clusters=k,

       random_state=42,

       n_init=10

   )

   kmeans.fit(X_scaled)

wcss.append(kmeans.inertia_)

# Plot Elbow Graph

plt.figure(figsize=(8, 5))

plt.plot(

   range(1, 11),

wcss,

   marker='o',

   linewidth=2

)

plt.title("Elbow Method for Optimal K")

plt.xlabel("Number of Clusters (K)")

plt.ylabel("WCSS")

plt.xticks(range(1, 11))

plt.grid(True)

plt.tight_layout()

plt.show()

# STEP 5: Implement K-Means Clustering

# Select 3 clusters

kmeans = KMeans(

   n_clusters=3,

   random_state=42,

   n_init=10

)

kmeans_labels = kmeans.fit_predict(X_scaled)

df["KMeans_Cluster"] = kmeans_labels

print("\nK-Means Cluster Counts:")

print(

   df["KMeans_Cluster"]

   .value_counts()

   .sort_index()

)

# STEP 6: Visualize K-Means Clusters using PCA

pca = PCA(n_components=2)

X_pca = pca.fit_transform(X_scaled)

plt.figure(figsize=(8, 5))

plt.scatter(

   X_pca[:, 0],

   X_pca[:, 1],

   c=kmeans_labels,

   cmap="viridis",

   s=50

)

plt.title("K-Means Clustering of Iris Dataset")

plt.xlabel("Principal Component 1")

plt.ylabel("Principal Component 2")

plt.colorbar(label="Cluster")

plt.grid(True)

plt.tight_layout()

plt.show()

# STEP 7: Implement Hierarchical Clustering

hierarchical = AgglomerativeClustering(

   n_clusters=3,

   linkage="ward"

)

hierarchical_labels = hierarchical.fit_predict(X_scaled)

df["Hierarchical_Cluster"] = hierarchical_labels

print("\nHierarchical Cluster Counts:")

print(

   df["Hierarchical_Cluster"]

   .value_counts()

   .sort_index()

)

# STEP 8: Plot Dendrogram

Z = linkage(

   X_scaled,

   method="ward"

)

plt.figure(figsize=(12, 6))

dendrogram(

   Z,

   truncate_mode="lastp",

   p=30,

   show_leaf_counts=True

)

plt.title("Dendrogram - Hierarchical Clustering")

plt.xlabel("Data Points / Merged Groups")

plt.ylabel("Ward Linkage Distance")

plt.tight_layout()

plt.show()

# STEP 9: Visualize Hierarchical Clustering

plt.figure(figsize=(8, 5))

plt.scatter(

   X_pca[:, 0],

   X_pca[:, 1],

   c=hierarchical_labels,

   cmap="plasma",

   s=50

)

plt.title("Hierarchical Clustering of Iris Dataset")

plt.xlabel("Principal Component 1")

plt.ylabel("Principal Component 2")

plt.colorbar(label="Cluster")

plt.grid(True)

plt.tight_layout()

plt.show()

# STEP 10: Compare Both Clustering Algorithms

kmeans_silhouette = silhouette_score(

   X_scaled,

   kmeans_labels

)

hierarchical_silhouette = silhouette_score(

   X_scaled,
